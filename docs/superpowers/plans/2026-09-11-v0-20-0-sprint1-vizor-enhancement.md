**Goal:** Directly enhance Synlynk Vizor (`synlynk viz`) to support the BS-6 Tri-Directional Visualizer (Product / Logical / Infra views), a streamlined Workspace Agent Roles & Onboarding Studio, an interactive Worktree Lifecycle & Sweep panel in Observatory, and full multi-agent harness synchronization (including Meta Muse).

**Architecture:** Extend `synlynk/viz.py` by introducing a modular projection engine (`synlynk/viz_views.py`) that materializes Product, Logical, and Infra nodes and edges into SQLite (`state.db`). Upgrade Vizor's master navigation shell to render self-contained, offline-ready HTML views (`product.html`, `logical.html`, `infra.html`, `roles.html`) with cross-view linking, while embedding worktree verdict diagnostics into `observatory.html` with safe 1-click cleanup actions.

**Tech Stack:** Python 3.10+, SQLite3 (stdlib), HTML5, Vanilla JavaScript, CSS3 (Obsidian Kinetic design tokens, zero external CDN dependencies).

**Specs:** 
- `docs/superpowers/specs/2026-09-11-v0.20.0-visual-workspace-autonomous-onboarding-design.md`
- `docs/superpowers/specs/2026-09-09-bs6-repo-workspace-visualization-design.md`

## Global Constraints
- **Zero External CDN Runtime Invariant:** All HTML views generated by Vizor must be self-contained (inline SVG, inline CSS, stdlib vanilla JS) to run in fully offline, air-gapped dev environments.
- **Local HTTP Auth:** All mutating POST endpoints must enforce `X-Synlynk-Token` authentication via `synlynk.local_http_auth.authorize_local_request`.
- **State Integrity:** Canonical state mutations occur in `state.db`; `.synlynk/viz-cache/` serves solely as disposable write-through presentation cache.
- **Backward Compatibility:** All existing 68 tests in `tests/test_viz*.py` must continue to pass; `journeys.html` requests must transparently redirect to `product.html`.

```python
# synlynk/viz_views.py
import json
import os
import re
import sqlite3
import time
from typing import Dict, List, Optional, Tuple

def init_workspace_view_tables(conn: sqlite3.Connection) -> None:
    cur = conn.cursor()
    cur.execute("""
    CREATE TABLE IF NOT EXISTS workspace_view_nodes (
        id TEXT PRIMARY KEY,
        view TEXT NOT NULL,
        repo TEXT NOT NULL,
        kind TEXT NOT NULL,
        label TEXT NOT NULL,
        attrs_json TEXT NOT NULL DEFAULT '{}',
        provenance TEXT NOT NULL,
        source_path TEXT,
        scanned_at TEXT NOT NULL,
        head_sha TEXT
    );
    """)
    cur.execute("""
    CREATE TABLE IF NOT EXISTS workspace_view_edges (
        id TEXT PRIMARY KEY,
        view TEXT NOT NULL,
        from_id TEXT NOT NULL,
        to_id TEXT NOT NULL,
        kind TEXT NOT NULL,
        provenance TEXT NOT NULL,
        attrs_json TEXT NOT NULL DEFAULT '{}',
        scanned_at TEXT NOT NULL,
        FOREIGN KEY (from_id) REFERENCES workspace_view_nodes(id),
        FOREIGN KEY (to_id) REFERENCES workspace_view_nodes(id)
    );
    """)
    cur.execute("""
    CREATE TABLE IF NOT EXISTS workspace_view_meta (
        view TEXT PRIMARY KEY,
        generated_at TEXT NOT NULL,
        head_sha TEXT,
        source_counts_json TEXT NOT NULL DEFAULT '{}',
        duration_ms INTEGER NOT NULL DEFAULT 0,
        stale INTEGER NOT NULL DEFAULT 0
    );
    """)
    conn.commit()

def extract_product_nodes(conn: sqlite3.Connection, repo_path: str) -> Tuple[List[dict], List[dict]]:
    nodes = []
    edges = []
    now = time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())
    repo_name = os.path.basename(os.path.abspath(repo_path)) or "repo"
    
    journeys_dir = os.path.join(repo_path, "docs", "journeys")
    if os.path.isdir(journeys_dir):
        for fname in sorted(os.listdir(journeys_dir)):
            if not fname.endswith(".md"):
                continue
            jpath = os.path.join(journeys_dir, fname)
            j_id = f"product:{repo_name}:journey:{os.path.splitext(fname)[0]}"
            nodes.append({
                "id": j_id, "view": "product", "repo": repo_name, "kind": "journey",
                "label": os.path.splitext(fname)[0].replace("-", " ").title(),
                "attrs_json": json.dumps({"file": f"docs/journeys/{fname}"}),
                "provenance": "authored", "source_path": f"docs/journeys/{fname}",
                "scanned_at": now, "head_sha": ""
            })
            try:
                with open(jpath) as f:
                    for line in f:
                        if line.startswith("route:"):
                            route = line.split(":", 1)[1].strip()
                            r_id = f"product:{repo_name}:route:{route.replace('/', '_')}"
                            nodes.append({
                                "id": r_id, "view": "product", "repo": repo_name, "kind": "route",
                                "label": route, "attrs_json": json.dumps({"route": route}),
                                "provenance": "extracted", "source_path": f"docs/journeys/{fname}",
                                "scanned_at": now, "head_sha": ""
                            })
                            edges.append({
                                "id": f"edge:{j_id}->{r_id}", "view": "product",
                                "from_id": j_id, "to_id": r_id, "kind": "includes",
                                "provenance": "authored", "attrs_json": "{}", "scanned_at": now
                            })
            except Exception:
                pass
    return nodes, edges

def extract_logical_nodes(conn: sqlite3.Connection, repo_path: str) -> Tuple[List[dict], List[dict]]:
    nodes = []
    edges = []
    now = time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())
    repo_name = os.path.basename(os.path.abspath(repo_path)) or "repo"
    
    known_pkgs = ["synlynk", "website", "tests", "bin"]
    for pkg in known_pkgs:
        p_path = os.path.join(repo_path, pkg)
        if os.path.isdir(p_path):
            pkg_id = f"logical:{repo_name}:package:{pkg}"
            nodes.append({
                "id": pkg_id, "view": "logical", "repo": repo_name, "kind": "package",
                "label": pkg, "attrs_json": json.dumps({"dir": pkg}),
                "provenance": "extracted", "source_path": pkg,
                "scanned_at": now, "head_sha": ""
            })
            for root, _, files in os.walk(p_path):
                for f in files:
                    if f.endswith(".py") and not f.startswith("."):
                        rel = os.path.relpath(os.path.join(root, f), repo_path)
                        mod_id = f"logical:{repo_name}:module:{rel}"
                        nodes.append({
                            "id": mod_id, "view": "logical", "repo": repo_name, "kind": "module",
                            "label": f, "attrs_json": json.dumps({"path": rel}),
                            "provenance": "extracted", "source_path": rel,
                            "scanned_at": now, "head_sha": ""
                        })
                        edges.append({
                            "id": f"edge:{pkg_id}->{mod_id}", "view": "logical",
                            "from_id": pkg_id, "to_id": mod_id, "kind": "includes",
                            "provenance": "extracted", "attrs_json": "{}", "scanned_at": now
                        })
    return nodes, edges

def extract_infra_nodes(conn: sqlite3.Connection, repo_path: str) -> Tuple[List[dict], List[dict]]:
    nodes = []
    edges = []
    now = time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())
    repo_name = os.path.basename(os.path.abspath(repo_path)) or "repo"
    
    daemon_id = f"infra:{repo_name}:service:daemon"
    nodes.append({
        "id": daemon_id, "view": "infra", "repo": repo_name, "kind": "service",
        "label": "Synlynk Daemon (Background)", "attrs_json": json.dumps({"type": "process"}),
        "provenance": "extracted", "source_path": ".synlynk/daemon.pid",
        "scanned_at": now, "head_sha": ""
    })
    vizor_id = f"infra:{repo_name}:service:vizor"
    nodes.append({
        "id": vizor_id, "view": "infra", "repo": repo_name, "kind": "service",
        "label": "Vizor Server (:8721)", "attrs_json": json.dumps({"port": 8721}),
        "provenance": "extracted", "source_path": "synlynk/viz.py",
        "scanned_at": now, "head_sha": ""
    })
    edges.append({
        "id": f"edge:{daemon_id}->{vizor_id}", "view": "infra",
        "from_id": daemon_id, "to_id": vizor_id, "kind": "manages",
        "provenance": "inferred", "attrs_json": "{}", "scanned_at": now
    })
    return nodes, edges

def build_workspace_views_snapshot(conn: sqlite3.Connection, repo_path: str) -> dict:
    init_workspace_view_tables(conn)
    p_nodes, p_edges = extract_product_nodes(conn, repo_path)
    l_nodes, l_edges = extract_logical_nodes(conn, repo_path)
    i_nodes, i_edges = extract_infra_nodes(conn, repo_path)
    
    return {
        "product": {"nodes": p_nodes, "edges": p_edges},
        "logical": {"nodes": l_nodes, "edges": l_edges},
        "infra": {"nodes": i_nodes, "edges": i_edges},
        "updated_at": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())
    }
```
