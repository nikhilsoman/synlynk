```python
def _scan_stage_stack(root: str, results: dict) -> None:
    """Stage 1 — Detect language, version, frameworks, CI, dep count, lockfile freshness."""
    import fnmatch as _fnmatch

    # ── Language detection ────────────────────────────────────────────────
    lang_markers = {
        "python": ["*.py", "pyproject.toml", "setup.py", "requirements.txt"],
        "node": ["package.json"],
        "go": ["go.mod"],
        "ruby": ["Gemfile"],
    }
    language = "unknown"
    for lang, patterns in lang_markers.items():
        for pat in patterns:
            if "*" in pat:
                found = any(
                    _fnmatch.fnmatch(fn, pat)
                    for dirpath, _, filenames in os.walk(root)
                    if ".git" not in dirpath and "__pycache__" not in dirpath
                    for fn in filenames
                )
            else:
                found = os.path.exists(os.path.join(root, pat))
            if found:
                language = lang
                break
        if language != "unknown":
            break

    # ── Version detection ─────────────────────────────────────────────────
    version = "unknown"
    for vfile, pat in [
        (".python-version", None),
        ("pyproject.toml", r'python_requires\s*[=<>!]+\s*["\']([^"\']+)'),
        (".nvmrc", None),
        (".node-version", None),
        ("go.mod", r"^go\s+(\S+)"),
    ]:
        vpath = os.path.join(root, vfile)
        if not os.path.exists(vpath):
            continue
        try:
            content = open(vpath, errors="ignore").read()
            if pat:
                m = re.search(pat, content, re.MULTILINE)
                if m:
                    version = m.group(1)
            else:
                first = content.strip().splitlines()
                if first:
                    version = first[0].strip()
            if version != "unknown":
                break
        except OSError:
            pass

    # ── Framework detection ───────────────────────────────────────────────
    frameworks = []
    fw_checks = [
        ("pytest", "pyproject.toml", "pytest"),
        ("pytest", "pytest.ini", None),
        ("pytest", "setup.cfg", "pytest"),
        ("django", "manage.py", None),
        ("flask", "requirements.txt", "flask"),
        ("fastapi", "requirements.txt", "fastapi"),
        ("next", "next.config.js", None),
        ("next", "next.config.ts", None),
        ("react", "package.json", "react"),
    ]
    seen_fw = set()
    for fw, fname, marker in fw_checks:
        if fw in seen_fw:
            continue
        fpath = os.path.join(root, fname)
        if not os.path.exists(fpath):
            continue
        if marker is None:
            seen_fw.add(fw)
            frameworks.append(fw)
        else:
            try:
                if marker.lower() in open(fpath, errors="ignore").read().lower():
                    seen_fw.add(fw)
                    frameworks.append(fw)
            except OSError:
                pass

    # ── Package manager ───────────────────────────────────────────────────
    pm_candidates = ["pyproject.toml", "setup.py", "package.json", "go.mod",
                     "Gemfile", "requirements.txt"]
    package_manager = next(
        (f for f in pm_candidates if os.path.exists(os.path.join(root, f))),
        "unknown",
    )

    # ── CI detection ──────────────────────────────────────────────────────
    ci = (
        os.path.isdir(os.path.join(root, ".github", "workflows"))
        or os.path.exists(os.path.join(root, ".gitlab-ci.yml"))
        or os.path.isdir(os.path.join(root, ".circleci"))
    )
    ci_workflows = 0
    wf_dir = os.path.join(root, ".github", "workflows")
    if os.path.isdir(wf_dir):
        ci_workflows = sum(
            1 for f in os.listdir(wf_dir)
            if f.endswith((".yml", ".yaml"))
        )

    # ── Dependency count ──────────────────────────────────────────────────
    dep_count = {"prod": 0, "dev": 0}
    req_path = os.path.join(root, "requirements.txt")
    if os.path.exists(req_path):
        try:
            dep_count["prod"] = sum(
                1 for line in open(req_path, errors="ignore")
                if line.strip() and not line.startswith("#")
            )
        except OSError:
            pass
    req_dev = os.path.join(root, "requirements-dev.txt")
    if os.path.exists(req_dev):
        try:
            dep_count["dev"] = sum(
                1 for line in open(req_dev, errors="ignore")
                if line.strip() and not line.startswith("#")
            )
        except OSError:
            pass
    pkg_json = os.path.join(root, "package.json")
    if os.path.exists(pkg_json):
        try:
            import json as _j
            data = _j.loads(open(pkg_json, errors="ignore").read())
            dep_count["prod"] = len(data.get("dependencies", {}))
            dep_count["dev"] = len(data.get("devDependencies", {}))
        except (OSError, ValueError):
            pass

    # ── Lockfile freshness ────────────────────────────────────────────────
    lockfile_pairs = [
        ("package.json", "package-lock.json"),
        ("package.json", "yarn.lock"),
        ("Gemfile", "Gemfile.lock"),
        ("go.mod", "go.sum"),
    ]
    lockfile_fresh = False
    for manifest, lockfile in lockfile_pairs:
        mpath = os.path.join(root, manifest)
        lpath = os.path.join(root, lockfile)
        if os.path.exists(mpath) and os.path.exists(lpath):
            lockfile_fresh = os.path.getmtime(lpath) >= os.path.getmtime(mpath)
            break
    if os.path.exists(os.path.join(root, "pyproject.toml")):
        lockfile_fresh = True  # pyproject.toml is self-describing

    results["stack"] = {
        "language": language,
        "version": version,
        "frameworks": frameworks,
        "package_manager": package_manager,
        "ci": ci,
        "ci_workflows": ci_workflows,
        "dep_count": dep_count,
        "lockfile_fresh": lockfile_fresh,
    }
```

```python
def _scan_stage_source(root: str, results: dict) -> None:
    """Stage 2 — AST-parse every .py file; per-file metrics sorted by lines desc."""
    import ast as _ast

    _SKIP = (".git", "__pycache__", "node_modules", ".venv", "venv", ".tox")

    def _should_skip(dirpath: str) -> bool:
        return any(skip in dirpath.split(os.sep) for skip in _SKIP)

    file_results = []
    for dirpath, _, filenames in os.walk(root):
        if _should_skip(dirpath):
            continue
        for fn in filenames:
            if not fn.endswith(".py"):
                continue
            filepath = os.path.join(dirpath, fn)
            rel_path = os.path.relpath(filepath, root)
            try:
                with open(filepath, encoding="utf-8", errors="ignore") as fh:
                    content = fh.read()
            except OSError:
                continue

            lines = content.count("\n") + 1

            try:
                tree = _ast.parse(content, filename=filepath)
            except SyntaxError:
                file_results.append({
                    "path": rel_path, "lines": lines, "functions": 0,
                    "classes": 0, "typed_pct": 0, "docstring_pct": 0,
                    "largest_fns": [], "parse_error": True,
                })
                continue

            all_fns = [n for n in _ast.walk(tree) if isinstance(n, _ast.FunctionDef)]
            classes = [n for n in _ast.walk(tree) if isinstance(n, _ast.ClassDef)]
            public_fns = [f for f in all_fns if not f.name.startswith("_")]

            typed = sum(
                1 for f in all_fns
                if f.returns or any(a.annotation for a in f.args.args)
            )
            typed_pct = int(typed * 100 / len(all_fns)) if all_fns else 0

            with_doc = sum(
                1 for f in public_fns if _ast.get_docstring(f)
            )
            docstring_pct = int(with_doc * 100 / len(public_fns)) if public_fns else 0

            fn_sizes = sorted(
                [
                    {
                        "name": f.name,
                        "lines": getattr(f, "end_lineno", f.lineno) - f.lineno,
                        "lineno": f.lineno,
                    }
                    for f in all_fns
                ],
                key=lambda x: x["lines"],
                reverse=True,
            )

            file_results.append({
                "path": rel_path,
                "lines": lines,
                "functions": len(all_fns),
                "classes": len(classes),
                "typed_pct": typed_pct,
                "docstring_pct": docstring_pct,
                "largest_fns": fn_sizes[:3],
            })

    file_results.sort(key=lambda x: x["lines"], reverse=True)
    results["source"] = file_results
```

```python
def _scan_stage_complexity(root: str, results: dict) -> None:
    """Stage 3 — Hotspot detection (functions >50 lines, files >500 lines) + TODO counts."""
    import ast as _ast

    _SKIP = (".git", "__pycache__", "node_modules", ".venv", "venv", ".tox")

    hotspots = []
    todo_counts = {"TODO": 0, "FIXME": 0, "HACK": 0, "XXX": 0}

    for dirpath, _, filenames in os.walk(root):
        if any(skip in dirpath.split(os.sep) for skip in _SKIP):
            continue
        for fn in filenames:
            if not fn.endswith(".py"):
                continue
            filepath = os.path.join(dirpath, fn)
            rel_path = os.path.relpath(filepath, root)
            try:
                with open(filepath, encoding="utf-8", errors="ignore") as fh:
                    content = fh.read()
            except OSError:
                continue

            for marker in todo_counts:
                todo_counts[marker] += len(re.findall(rf"\b{marker}\b", content))

            lines = content.count("\n") + 1
            if lines > 500:
                hotspots.append({

                        "path": rel_path,
                        "fn": node.name,
                        "lines": fn_lines,
                        "lineno": node.lineno,
                    })

    hotspots.sort(key=lambda x: x["lines"], reverse=True)
```

```python
def _scan_stage_tests(root: str, results: dict) -> None:
    """Stage 4 — Structural name-matching coverage: which public fns have no test."""
    import ast as _ast

    _SKIP = (".git", "__pycache__", "node_modules", ".venv", "venv", ".tox")

    source_fns = []   # {"name": str, "file": str, "lineno": int}
    test_covered = set()  # function names that appear to be tested

    for dirpath, _, filenames in os.walk(root):
        if any(skip in dirpath.split(os.sep) for skip in _SKIP):
            continue
        for fn in filenames:
            if not fn.endswith(".py"):
                continue
            filepath = os.path.join(dirpath, fn)
            rel_path = os.path.relpath(filepath, root)
            is_test = fn.startswith("test_") or fn.endswith("_test.py")
            try:
                with open(filepath, encoding="utf-8", errors="ignore") as fh:
                    content = fh.read()
                tree = _ast.parse(content, filename=filepath)
            except (OSError, SyntaxError):
                continue

            if is_test:
                for node in _ast.walk(tree):
                    if isinstance(node, _ast.FunctionDef) and node.name.startswith("test_"):
                        candidate = node.name[5:]  # strip "test_" prefix
                        test_covered.add(candidate)
                    # also collect direct call targets inside test functions
                    elif isinstance(node, _ast.Call):
                        if isinstance(node.func, _ast.Name):
                            test_covered.add(node.func.id)
                        elif isinstance(node.func, _ast.Attribute):
                            test_covered.add(node.func.attr)
            else:
                for node in _ast.walk(tree):
                    if isinstance(node, _ast.FunctionDef) and not node.name.startswith("_"):
                        source_fns.append({
                            "name": node.name,
                            "file": rel_path,
                            "lineno": node.lineno,
                        })

    covered = [f for f in source_fns if f["name"] in test_covered]
    gaps = [f for f in source_fns if f["name"] not in test_covered]
    total = len(source_fns)
    results["tests"] = {
        "gap_functions": gaps,
        "covered_count": len(covered),
        "gap_count": len(gaps),
        "ratio": len(covered) / total if total > 0 else 0.0,
    }
```

```python
def _scan_stage_git(root: str, results: dict) -> None:
    """Stage 5 — Git churn: most-changed files in last 30 commits."""
    import datetime as _dt

    error_result = {"error": "", "churn": [], "total_commits_scanned": 0}

    try:
        proc = subprocess.run(
            ["git", "log", "--name-only", "--pretty=format:COMMIT:%ai", "-n", "30"],
            cwd=root,
            capture_output=True,
            text=True,
            timeout=5,
        )
        if proc.returncode != 0:
            error_result["error"] = f"git log failed: {proc.stderr.strip()[:120]}"
            results["git"] = error_result
            return
    except (subprocess.TimeoutExpired, FileNotFoundError, OSError) as exc:
        error_result["error"] = str(exc)[:120]
        results["git"] = error_result
        return

    file_counts: dict = {}
    last_date: dict = {}
    commit_count = 0
    current_date = None

    for line in proc.stdout.splitlines():
        line = line.strip()
        if line.startswith("COMMIT:"):
            commit_count += 1
            try:
                # "COMMIT:2026-07-03 14:22:11 +0000" — take first 19 chars of date part
                date_str = line[7:26]
                current_date = _dt.datetime.fromisoformat(date_str)
            except ValueError:
                current_date = None
        elif line:
            file_counts[line] = file_counts.get(line, 0) + 1
            if current_date and line not in last_date:
                last_date[line] = current_date

    now = _dt.datetime.now()
    churn = []
    for path, count in file_counts.items():
        last = last_date.get(path)
        days_ago = (now - last).days if last else 0
        if count > 20:
            temp = "hot"
        elif count >= 5:
            temp = "warm"
        else:
            temp = "cold"
        churn.append({
            "path": path,
            "commits": count,
            "last_days_ago": days_ago,
            "temp": temp,
        })

    churn.sort(key=lambda x: x["commits"], reverse=True)
    results["git"] = {"churn": churn, "total_commits_scanned": commit_count}
```

```python
def _scan_stage_arch(root: str, results: dict) -> None:
    """Stage 6 — Entry points, import graph, dead code candidates, pattern classification."""
    import ast as _ast

    _SKIP = (".git", "__pycache__", "node_modules", ".venv", "venv", ".tox")
    pkg_name = os.path.basename(root)

    entry_points = []
    import_graph: dict = {}
    inbound: dict = {}
    file_line_counts: dict = {}
    public_count = 0

    for dirpath, _, filenames in os.walk(root):
        if any(skip in dirpath.split(os.sep) for skip in _SKIP):
            continue
        for fn in filenames:
            if not fn.endswith(".py"):
                continue
            filepath = os.path.join(dirpath, fn)
            rel_path = os.path.relpath(filepath, root)
            is_test = os.path.basename(fn).startswith("test_")
            try:
                with open(filepath, encoding="utf-8", errors="ignore") as fh:
                    content = fh.read()
            except OSError:
                continue

            lines = content.count("\n") + 1
            file_line_counts[rel_path] = lines

            try:
                tree = _ast.parse(content, filename=filepath)
            except SyntaxError:
                continue

            if not is_test:
                for node in _ast.walk(tree):
                    if isinstance(node, (_ast.FunctionDef, _ast.ClassDef)):
                        if not node.name.startswith("_"):
                            public_count += 1

            # Entry points
            for node in _ast.walk(tree):
                if isinstance(node, _ast.FunctionDef) and node.name == "main":
                    entry_points.append({
                        "name": "main", "file": rel_path, "lineno": node.lineno,
                    })
                elif isinstance(node, _ast.If):
                    test = node.test
                    if (isinstance(test, _ast.Compare)
                            and isinstance(test.left, _ast.Name)
                            and test.left.id == "__name__"
                            and any(
                                isinstance(c, _ast.Constant) and c.value == "__main__"
                                for c in test.comparators
                            )):
                        entry_points.append({
                            "name": "__main__", "file": rel_path, "lineno": node.lineno,
                        })

            # Local import graph
            local_imports = []
            for node in _ast.walk(tree):
                if isinstance(node, _ast.ImportFrom):
                    if node.level > 0:
                        # relative import — same package
                        local_imports.append(rel_path)
                    elif node.module and node.module.split(".")[0] == pkg_name:
                        mod_path = node.module.replace(".", os.sep) + ".py"
                        local_imports.append(mod_path)
                        inbound[mod_path] = inbound.get(mod_path, 0) + 1
            if local_imports:
                import_graph[rel_path] = local_imports

    total_lines = sum(file_line_counts.values()) or 1
    max_lines = max(file_line_counts.values(), default=0)

    if max_lines / total_lines > 0.5:
        pattern = "monolith"
    elif not entry_points:
        pattern = "library"
    else:
        pattern = "modular"

    source_files = {
        p for p in file_line_counts
        if not os.path.basename(p).startswith("test_")
    }
    dead_candidates = sorted(
        f for f in source_files
        if inbound.get(f, 0) == 0 and "__init__" not in f
    )

    results["arch"] = {
        "entry_points": entry_points,
        "import_graph": import_graph,
        "dead_candidates": dead_candidates,
        "public_api_count": public_count,
        "pattern": pattern,
    }
```

```python
def _kbhit() -> bool:
    """Return True if a keypress is waiting on stdin (non-blocking)."""
    import select as _select
    if not sys.stdin.isatty():
        return False
    try:
        ready, _, _ = _select.select([sys.stdin], [], [], 0)
        return bool(ready)
    except Exception:
        return False
```

```python
def _run_scan_tui(results: dict, threads: list, primary_root: str = ".") -> None:
    """Poll stage results every 200ms, render Stage Cards, handle keypresses."""
    import time as _time

    expanded = None
    scan_start = _time.monotonic()

    while True:
        elapsed = _time.monotonic() - scan_start
        all_done = all(results.get(k) is not None for k in STAGE_KEYS)

        _wiz_clear()
        _render_scan_cards(results, expanded, elapsed)

        if _kbhit():
            key = _wiz_read_key()
            if key.isdigit() and 1 <= int(key) <= 6:
                new_exp = STAGE_KEYS[int(key) - 1]
                expanded = None if expanded == new_exp else new_exp
            elif key in ("\r", "\n"):
                if all_done:
                    break
            elif key == "r":
                # Re-scan: caller handles this by re-invoking; just exit TUI
                return
            elif key in ("q", "\x03"):
                sys.exit(0)

        if all_done and not _kbhit():
            # Auto-proceed after brief pause if no interaction
            pass  # wait for explicit [enter]; don't auto-proceed

        _time.sleep(0.2)

    # ── Completion sequence ───────────────────────────────────────────────
    import json as _json
    updated = _write_scan_fences(results, root=primary_root)
    if updated:
        print(f"\n  {_GREEN}── Agent fences updated ──────────────────{_RESET}")
        for path in updated:
            print(f"  {_GREEN}✓{_RESET} {path}  — codebase context written")

    scan_json = os.path.join(primary_root, ".synlynk", "scan-result.json")
    os.makedirs(os.path.dirname(scan_json), exist_ok=True)
    try:
        open(scan_json, "w").write(_json.dumps(results, default=str, indent=2))
        print(f"  {_DIM}.synlynk/scan-result.json saved{_RESET}")
    except OSError:
        pass

    print(f"\n  {_DIM}Press{_RESET} {_CYAN}[enter]{_RESET} {_DIM}to run{_RESET} "
          f"{_CYAN}synlynk launch{_RESET}  {_DIM}or [q] to quit{_RESET}")
    key = _wiz_read_key()
    if key in ("\r", "\n"):
        cmd_launch_ftue()
    elif key in ("q", "\x03"):
        sys.exit(0)
```

```python
    # ── Complexity Hotspots section ───────────────────────────────────────
    hotspots = complexity.get("hotspots", [])
    if hotspots:
        body_lines.append("")
        body_lines.append("## Complexity Hotspots")
        for h in hotspots[:3]:
            fn_label = f"{h['fn']}()" if h.get("fn") else os.path.basename(h["path"])
            body_lines.append(
                f"- {fn_label} — {h['lines']} lines ({h['path']}:{h['lineno']})"
            )

    # ── Test Gaps section ─────────────────────────────────────────────────
    gap_count = tests_data.get("gap_count", 0)
    if gap_count > 0:
        body_lines.append("")
        body_lines.append("## Test Gaps (structural, not runtime coverage)")
        gap_names = [g["name"] for g in tests_data.get("gap_functions", [])[:5]]
        suffix = f" [+{gap_count - 5} more]" if gap_count > 5 else ""
        body_lines.append(
            f"- {gap_count} untested public function{'s' if gap_count != 1 else ''}: "
            f"{', '.join(gap_names)}{suffix}"
        )

    # ── Hot Files section ─────────────────────────────────────────────────
    churn = git_data.get("churn", [])
    has_git_error = "error" in git_data and not churn
    if churn and not has_git_error:
        body_lines.append("")
        body_lines.append("## Hot Files (last 30 commits)")
        for c in churn[:3]:
            icon = "🔥" if c["temp"] == "hot" else ("⚡" if c["temp"] == "warm" else "·")
            body_lines.append(
                f"- {icon} {c['path']} — {c['commits']} commits"
            )

    # ── Tech Debt section ─────────────────────────────────────────────────
    todo_counts = complexity.get("todo_counts", {})
    total_markers = sum(todo_counts.values())
    if total_markers > 0:
        body_lines.append("")
        body_lines.append("## Tech Debt")
        parts = [f"{v} {k}" for k, v in todo_counts.items() if v]
        body_lines.append(f"- {' · '.join(parts)}")

    body = "\n".join(body_lines)
    scan_date = _dt.date.today().isoformat()

    updated = []
    for fname in directive_files:
        fpath = os.path.join(root, fname)
        if not os.path.exists(fpath):
            continue
        _upsert_harness_fence(fpath, f"scan-{scan_date}", body)
        updated.append(fpath)

    return updated
```
