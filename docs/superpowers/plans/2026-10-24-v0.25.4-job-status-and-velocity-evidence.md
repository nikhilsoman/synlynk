"""One-off audit: list cost_entries rows whose harness/model disagree with
their daemon_jobs.actual_harness/resolved_model. Report only -- does not
modify any row. gh:#2547."""
import sys
sys.path.insert(0, "synlynk")  # noqa: E402
import synlynk


def main():
    conn = synlynk._get_db()
    rows = conn.execute(
        """SELECT ce.id, ce.job_id, ce.harness, ce.model,
                  dj.actual_harness, dj.resolved_model
           FROM cost_entries ce
           JOIN daemon_jobs dj ON ce.job_id = dj.job_id
           WHERE dj.actual_harness IS NOT NULL
             AND ce.harness IS NOT NULL
             AND ce.harness != dj.actual_harness"""
    ).fetchall()
    conn.close()
    if not rows:
        print("No harness-attribution mismatches found.")
        return 0
    print(f"{len(rows)} mismatched cost_entries rows:")
    for r in rows:
        print(f"  cost_entries.id={r[0]} job_id={r[1]} "
              f"cost_entries.harness={r[2]!r} vs daemon_jobs.actual_harness={r[4]!r} "
              f"(model {r[3]!r} vs {r[5]!r})")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

def _record_job_timing_metrics(conn, job_id: str, completed_at: str) -> None:
    row = conn.execute(
        "SELECT actual_harness, resolved_model, requested_harness, "
        "enqueued_at, started_at FROM daemon_jobs WHERE job_id=?",
        (job_id,),
    ).fetchone()
    if not row:
        return
    harness, model, _requested, enqueued_at, started_at = row

    def _parse(ts):
        return time.mktime(time.strptime(ts, "%Y-%m-%dT%H:%M:%S")) if ts else None

    queue_s = None
    active_s = None
    t_enq, t_start, t_end = _parse(enqueued_at), _parse(started_at), _parse(completed_at)
    if t_enq is not None and t_start is not None:
        queue_s = t_start - t_enq
    if t_start is not None and t_end is not None:
        active_s = t_end - t_start

    self_reported = _extract_self_reported_timing(job_id)
    cross_check = "absent"
    if self_reported is not None:
        reported_active = self_reported.get("active_seconds")
        if reported_active is not None and active_s is not None:
            cross_check = "agree" if abs(reported_active - active_s) < 5 else "disagree"

    conn.execute(
        """INSERT INTO job_timing_metrics
           (job_id, harness, model, queue_seconds, active_seconds,
            self_reported_json, cross_check_status)
